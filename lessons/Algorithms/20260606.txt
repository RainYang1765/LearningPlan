class Solution:
    def topKstrings(self , strings , k ):
        # write code here
        dict={}
        for i  in range(len(strings)):
            if strings[i] not in dict:
                dict[strings[i]]=1
            else:
                dict[strings[i]] +=1
        return (sorted(dict.items(),key=lambda x:(-x[1],x[0]))[:k])
