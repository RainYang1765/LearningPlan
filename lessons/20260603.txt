BM35 判断是不是完全二叉树

class Solution:
    def isCompleteTree(self, root: TreeNode) -> bool:
        # write code here
        if root.left == root.right == None:
            return True
        #
        q:list[TreeNode]=[root]
        layer_idx=[0]
        node_list:list[list[str]]=[]
        while q:
            node=q.pop(0)
            layer=layer_idx.pop(0)
            if layer==len(node_list):
                node_list+=[[]]
            if node:
                node_list[layer].append(str(node.val))
                q.append(node.left)
                layer_idx.append(layer+1)
                q.append(node.right)
                layer_idx.append(layer+1)
            else:
                node_list[layer].append('#')
        #
        if all(node=='#' for node in node_list[-1]):
            node_list=node_list[:-1]
        #
        for i in range(len(node_list)):
            #
            if 0<i<len(node_list)-1:
                if len(node_list[i])!=2**i or ('#' in node_list[i]):
                    return False
            #
            if i==len(node_list)-1:
                for j in range(1,len(node_list[i])):
                    if node_list[i][j]!='#' and node_list[i][j-1]=='#':
                        return False
        return True
